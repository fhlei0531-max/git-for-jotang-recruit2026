# Git Bash & Git 操作


# 1. 基础 Bash 命令 
pwd                     # 查看你当前所在的绝对路径 (Print Working Directory)
ls                      # 列出当前文件夹下的所有文件和目录 (List)
ls -a                   # 列出所有文件，包括隐藏文件 (比如 .git 文件夹)
cd 文件夹名              # 进入某个文件夹 (Change Directory)
cd ..                   # 退回上一级目录
mkdir 项目名             # 创建一个新的文件夹 (Make Directory)
rm -rf 文件名/文件夹名    # 强制删除文件或文件夹 (警告：删了就找不回来了，慎用！)

# 2. Git 初始配置
git config --global user.name "你的名字"      # 告诉 Git 你是谁
git config --global user.email "你的邮箱"     # 告诉 Git 你的联系方式
git config --list                           # 查看当前的 Git 配置信息

# 3. 开启 Git 
git init                # 把当前普通的文件夹，变成 Git 可以管理的本地仓库 (会生成隐藏的 .git)
git clone 仓库链接       # 把网上的代码仓库 (比如 GitHub/Gitee) 下载到本地

# 4. 日常提交 (存取快照)
git status              # 【高频使用】查看当前仓库的状态 (看看哪些文件改了，哪些还没提交)
git add 文件名           # 把特定文件的修改放到“暂存区”
git add .               # 【高频使用】把当前目录下所有修改过的文件都放到“暂存区”
git commit -m "提交说明" # 把暂存区的内容正式提交到本地仓库，必须写清楚这次改了啥 (比如 "新增SVM训练代码")

# 5. 查看历史与撤销 
git log                 # 查看历史提交记录 (按 Q 键退出)
git log --oneline       # 只看简略版的一行历史记录
git checkout -- 文件名   # 撤销工作区某个文件的修改，让它回到上次提交的状态
git reset HEAD 文件名    # 把放进暂存区的文件拿出来 (不删除修改内容)

# 6. 分支管理 (多人协作或开发新功能) 
git branch              # 查看当前有哪些分支 (*号代表你当前所在的分支)
git branch 分支名        # 创建一个新分支
git checkout 分支名      # 切换到那个分支去
git checkout -b 分支名   # 创建并立刻切换到新分支 (最常用的快捷招式)
git merge 分支名         # 把指定分支的代码合并到当前所在的分支里

# 7. 远程仓库 (推送与拉取)
git remote -v                       # 查看当前连接了哪个远程仓库
git remote add origin 仓库链接       # 如果你是本地 git init 的，需要用这步绑定远程仓库
git pull origin 分支名               # 把远程仓库的最新代码拉取到本地并合并 (上班第一件事)
git push origin 分支名               # 把本地提交好的代码推送到远程仓库 (下班最后一件事)
